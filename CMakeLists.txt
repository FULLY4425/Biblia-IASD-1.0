cmake_minimum_required(VERSION 3.28)
project(obs-biblia VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(libobs CONFIG REQUIRED)
find_package(obs-frontend-api CONFIG REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Network Sql)
add_library(obs-biblia MODULE src/plugin.cpp src/bible.cpp)
target_link_libraries(obs-biblia PRIVATE OBS::libobs OBS::obs-frontend-api Qt6::Widgets Qt6::Network)
add_executable(biblia-generador WIN32 src/generator.cpp src/openlp.cpp src/bible.cpp)
target_link_libraries(biblia-generador PRIVATE Qt6::Widgets Qt6::Sql)
if(MSVC)
  target_compile_options(biblia-generador PRIVATE /utf-8 /Zc:__cplusplus)
endif()
install(TARGETS biblia-generador RUNTIME DESTINATION generador)
if(MSVC)
  target_compile_options(obs-biblia PRIVATE /utf-8 /W4 /Zc:__cplusplus)
else()
  target_compile_options(obs-biblia PRIVATE -Wall -Wextra -Wpedantic)
endif()
set_target_properties(obs-biblia PROPERTIES PREFIX "")
if(WIN32)
  install(TARGETS obs-biblia RUNTIME DESTINATION obs-plugins/64bit LIBRARY DESTINATION obs-plugins/64bit)
else()
  install(TARGETS obs-biblia LIBRARY DESTINATION lib/obs-plugins)
endif()
install(DIRECTORY data/ DESTINATION data/obs-plugins/obs-biblia)
include(CTest)
if(BUILD_TESTING)
  add_executable(bible-tests tests/bible-tests.cpp src/bible.cpp)
  target_include_directories(bible-tests PRIVATE src)
  target_link_libraries(bible-tests PRIVATE Qt6::Core)
  if(MSVC)
    target_compile_options(bible-tests PRIVATE /utf-8 /Zc:__cplusplus)
  endif()
  add_test(NAME references-and-lookup COMMAND bible-tests)
  add_executable(text-fit-tests tests/text-fit-tests.cpp src/bible.cpp)
  target_include_directories(text-fit-tests PRIVATE src)
  target_link_libraries(text-fit-tests PRIVATE Qt6::Gui)
  if(MSVC)
    target_compile_options(text-fit-tests PRIVATE /utf-8 /Zc:__cplusplus)
  endif()
  add_test(NAME complete-chapter-fit COMMAND text-fit-tests "${CMAKE_CURRENT_SOURCE_DIR}/data/bibles/rv1909.json")
  set_tests_properties(complete-chapter-fit PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=windows;QT_DEBUG_PLUGINS=1" TIMEOUT 60)
  add_executable(extension-tests tests/extension-tests.cpp src/openlp.cpp src/bible.cpp)
  target_include_directories(extension-tests PRIVATE src)
  target_link_libraries(extension-tests PRIVATE Qt6::Widgets Qt6::Sql)
  if(MSVC)
    target_compile_options(extension-tests PRIVATE /utf-8 /Zc:__cplusplus)
  endif()
  add_test(NAME slides-css-and-openlp COMMAND extension-tests)
  set_tests_properties(slides-css-and-openlp PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=windows" TIMEOUT 60)
endif()
