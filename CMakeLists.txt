cmake_minimum_required(VERSION 3.28)
project(obs-biblia VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(libobs CONFIG REQUIRED)
find_package(obs-frontend-api CONFIG REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets)
add_library(obs-biblia MODULE src/plugin.cpp src/bible.cpp)
target_link_libraries(obs-biblia PRIVATE OBS::libobs OBS::obs-frontend-api Qt6::Widgets)
if(MSVC)
  target_compile_options(obs-biblia PRIVATE /utf-8 /W4 /Zc:__cplusplus)
else()
  target_compile_options(obs-biblia PRIVATE -Wall -Wextra -Wpedantic)
endif()
set_target_properties(obs-biblia PROPERTIES PREFIX "")
if(WIN32)
  install(TARGETS obs-biblia RUNTIME DESTINATION obs-plugins/64bit LIBRARY DESTINATION obs-plugins/64bit)
else()
  install(TARGETS obs-biblia LIBRARY DESTINATION lib/obs-plugins)
endif()
install(DIRECTORY data/ DESTINATION data/obs-plugins/obs-biblia)
include(CTest)
if(BUILD_TESTING)
  add_executable(bible-tests tests/bible-tests.cpp src/bible.cpp)
  target_include_directories(bible-tests PRIVATE src)
  target_link_libraries(bible-tests PRIVATE Qt6::Core)
  if(MSVC)
    target_compile_options(bible-tests PRIVATE /utf-8 /Zc:__cplusplus)
  endif()
  add_test(NAME references-and-lookup COMMAND bible-tests)
endif()
